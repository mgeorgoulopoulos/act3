if(WIN32)
    add_executable(act3 WIN32 platform/windows_window.cpp)
    target_link_libraries(act3 PRIVATE opengl32 gdi32 user32 windowscodecs ole32)
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT act3)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(X11 REQUIRED)
    find_package(PNG REQUIRED)
    find_package(OpenGL REQUIRED COMPONENTS OpenGL GLX)
    add_executable(act3 platform/linux_window.cpp)
    target_link_libraries(act3 PRIVATE X11::X11 OpenGL::OpenGL OpenGL::GLX PNG::PNG)
else()
    message(FATAL_ERROR "ACT3 supports Windows and Linux.")
endif()

file(READ "${model_source}" model_code)
function(extract_model_method output start_marker end_marker)
    string(FIND "${model_code}" "${start_marker}" start)
    string(FIND "${model_code}" "${end_marker}" end)
    if(start LESS 0 OR end LESS_EQUAL start)
        message(FATAL_ERROR "Cannot locate historical method: ${start_marker}")
    endif()
    math(EXPR length "${end} - ${start}")
    string(SUBSTRING "${model_code}" ${start} ${length} method)
    set(${output} "${method}" PARENT_SCOPE)
endfunction()
